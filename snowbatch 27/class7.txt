SELECT FIRST_NAME,
       SALARY * 0.15 AS hra,
       SALARY * 0.05 AS pf,
       SALARY * 0.10 AS da,
       (SALARY + (SALARY * 0.15) + (SALARY * 0.10) - (SALARY * 0.05)) AS total_salary
FROM EMPLOYEES
ORDER BY total_salary DESC;
